1. 安装 certbot

# 阿里云服务器
sudo yum install certbot -y

2. 修改80端口映射

修改nginx的域名80映射,使证书签发网站访问生效

server {
    listen 80;
    server_name www.***.com;

    # 只针对 /.well-known/acme-challenge/ 路径，直接返回对应文件，不跳转
    location ^~ /.well-known/acme-challenge/ {
        root /work/projects/web;
        allow all;
        try_files $uri =404;
    }

    # 其他所有请求跳转到 HTTPS
    location / {
        return 301 https://$server_name$request_uri;
    }
}

3. 测试生成

# 模拟测试续期  --dry-run 是测试,不进行真正的续签
sudo certbot renew --dry-run

4. 编写脚本

# 创建文件
cd /work/projects/ca
touch manage_cert.sh

# 赋予权限
chmod +x /work/projects/ca/manage_cert_renew.sh
# 内容如下
#!/bin/bash

# 你的域名
DOMAIN="www.***.com"
# 证书最终存放目录
DEST_DIR="/work/projects/ca"
# 你的 nginx 网站根目录（webroot）
WEBROOT="/work/projects/web"
# nginx 执行路径
NGINX_BIN="/usr/local/nginx/sbin/nginx"
# 邮箱地址（注册证书用）
EMAIL="***@qq.com"

# 申请或续期证书，并复制到指定目录
function obtain_or_renew_cert() {
    echo "开始申请或续期证书..."
    certbot certonly --webroot -w "$WEBROOT" -d "$DOMAIN" --email "$EMAIL" --agree-tos --non-interactive --quiet
    
    if [ $? -ne 0 ]; then
        echo "证书申请或续期失败！"
        exit 1
    fi

    echo "申请或续期成功，复制证书文件..."

    mkdir -p "$DEST_DIR"
    cp /etc/letsencrypt/live/$DOMAIN/fullchain.pem $DEST_DIR/cert.pem
    cp /etc/letsencrypt/live/$DOMAIN/privkey.pem $DEST_DIR/cert.key

    echo "证书已更新到 $DEST_DIR"
}

# 重载 nginx
function reload_nginx() {
    echo "重载 nginx 配置..."
    $NGINX_BIN -s reload
    if [ $? -eq 0 ]; then
        echo "nginx 重载成功"
    else
        echo "nginx 重载失败"
    fi
}

# 主流程
obtain_or_renew_cert
reload_nginx

5. 创建自动续期脚本

# 创建文件
cd /work/projects/ca
touch auto_renew.sh
# 赋予权限
chmod +x /work/projects/ca/manage_cert_renew.sh
# 内容如下
#!/bin/bash
DOMAIN="www.***.com"
EMAIL="**@qq.com"
DEST_DIR="/work/projects/ca"
NGINX_BIN="/usr/local/nginx/sbin/nginx"

echo "开始自动续期证书..."
certbot renew --quiet

if [ $? -eq 0 ]; then
    echo "续期成功，复制证书文件..."
    mkdir -p "$DEST_DIR"
    cp /etc/letsencrypt/live/$DOMAIN/fullchain.pem $DEST_DIR/cert.pem
    cp /etc/letsencrypt/live/$DOMAIN/privkey.pem $DEST_DIR/cert.key

    echo "证书复制完成，重载 nginx ..."
    $NGINX_BIN -s reload
    echo "nginx 重载完成"
else
    echo "续期失败"
fi
# 或者使用以下手动模式
certbot certonly --webroot -w "/work/projects/web" -d "www.***.com" --email "**@qq.com" --agree-tos --non-interactive --quiet

# 默认会发放证书到该目录下
cd /etc/letsencrypt/live/www.***.com

6. 配置linux脚本

# 查看当前定时任务
crontab -l 
# 编辑当前定时任务
crontab -e
# 输入以下定时任务
0 3 * * * /bin/bash /work/projects/ca/auto_renew.sh >> /var/log/cert_renew.log 2>&1
