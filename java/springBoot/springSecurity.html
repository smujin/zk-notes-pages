一、springSecurity介绍

Spring Security 是一个功能强大且高度可定制的身份验证和访问控制框架。它是用于保护基于Spring的应用程序的实际标准。Spring Security 致力于为Java应用程序提供身份验证和授权。与所有Spring项目一样，Spring Security的真正强大之处在于可以轻松扩展以满足自定义要求

三、基础过滤器介绍

1、FilterSecurityInterceptor: 是一个方法级的权限过滤器，基本位于过滤链的最底部

2、ExceptionTranslationFilter: 是个异常处理器，用来处理在认证授权过程中抛出的异常

3、UsernamePasswordAuthenticationFilter: 对login的post请求做拦截,校验表单中的用户名和密码

4、所有的过滤器都会被 FilterChainProxy 进行管理 在其方法 doFilterInternal 中获取到所有的过滤器对象

3、重要配置类

@Configuration
@EnableWebSecurity
public class SecurityConfig extends WebSecurityConfigurerAdapter {

    @Autowired
    private UserDetailsService userDetailsService;

    //注入数据源(需要注入数据源)
    @Autowired
    private DataSource dataSource;
    //配置对象(记住我配置)
    @Bean
    private PersistentTokenRepository persistentTokenRepository() {
        JdbcTokenRepositoryImpl jdbcTokenRepository = new JdbcTokenRepositoryImpl();
        jdbcTokenRepository.setDataSource(dataSource);
        //启动的时候把表创建了
//        jdbcTokenRepository.setCreateTableOnStartup(true);
        return jdbcTokenRepository;
    }

    @Override
    protected void configure(AuthenticationManagerBuilder auth) throws Exception {
        auth.userDetailsService(userDetailsService).passwordEncoder(encoder());
    }

    @Bean
    public PasswordEncoder encoder() {
        return new BCryptPasswordEncoder();
    }

    @Override
    protected void configure(HttpSecurity http) throws Exception {
        //配置没有权限跳转访问自定义页面
        http.exceptionHandling().accessDeniedPage("/unauth.html");
        http.formLogin().loginPage("/login.html") //转到自己的登录页面
                .loginProcessingUrl("/user/login") //登录访问路径
                .defaultSuccessUrl("/test/security").permitAll() //登录成功后的页面
                .and().authorizeRequests().antMatchers("/","/test/security","/user/login").permitAll() //访问这些路径的时候不需要认证
                //当前登录的用户拥有admin的权限才能访问该路径
//                .antMatchers("/test/hello").hasAuthority("admins")
                //只要有其中一个权限就可以访问
//                .antMatchers("/test/hello").hasAnyAuthority("admins,role")
                //拥有role角色才能访问 该源码中校验角色必须以 ROLE_ 开头
//                .antMatchers("/test/hello").hasRole("role")
                .antMatchers("/test/hello").hasAnyRole("ROLE_juese")
                .anyRequest().authenticated()//所有请求都需要认证
                .and().rememberMe().tokenRepository(persistentTokenRepository()) //记住我功能需要
                .tokenValiditySeconds(60) //60秒内不用输入账号密码即可重复跳过认证,单位秒,记住我功能需要
                .userDetailsService(userDetailsService) //用于操作数据库,记住我功能需要
                .and().csrf().disable(); //关闭csrf的防护(跨站请求伪造)
        //退出登录
        http.logout().logoutUrl("/logout").logoutSuccessUrl("/index").permitAll();
    }
}

4、重要自定义配置数据库

@Service("userDetailsService")
public class MyUserDetailsService implements UserDetailsService {
    @Autowired
    private UserMapper userMapper;

    @Override
    public UserDetails loadUserByUsername(String s) throws UsernameNotFoundException {
        //调用数据库信息
        QueryWrapper<Users> queryWrapper = new QueryWrapper();
        queryWrapper.eq("username", s);
        Users user = userMapper.selectOne(queryWrapper);
        //判断
        if (user == null) { //数据库没有值,认证失败
            throw new UsernameNotFoundException("用户名不存在!");
        }
        List<GrantedAuthority> auths = AuthorityUtils.commaSeparatedStringToAuthorityList("管理员");
        return new User(user.getUsername(), new BCryptPasswordEncoder().encode(user.getPassword()), auths);
    }
}

@Component
public class DefaultPasswordEncoder implements PasswordEncoder {

    public DefaultPasswordEncoder() {
        this(-1);
    }

    public DefaultPasswordEncoder(int strength) {
    }

    @Override
    public String encode(CharSequence charSequence) {
        //对密码进行MD5加密
        return DigestUtils.md5DigestAsHex(charSequence.toString().getBytes());
    }

    @Override
    public boolean matches(CharSequence charSequence, String s) {
        //对加密密码和输入的密码进行比对 s是已经加密的
        return s.equals(DigestUtils.md5DigestAsHex(charSequence.toString().getBytes()));
    }
}

@Component
public class TokenManager {

    //设置token有效时长
    private final long tokenEcpiration = 24 * 60 * 60 * 1000;
    //编码秘钥
    private final String TOKEN_SIGN_KEY = "123456";

    //1. 根据用户名生成token
    public String createToken(String username) {
        // 生成jwt访问令牌
        String jwtToken = Jwts.builder()
                // 主题 - 存用户名
                .setSubject(username)
                // 用户角色
//                .claim("ROLE_LOGIN", "ADMIN")
                // 过期时间,这里可以优化一下时间
                .setExpiration(new Date(System.currentTimeMillis() + tokenEcpiration))
                // 加密算法和密钥
                .signWith(SignatureAlgorithm.HS512, TOKEN_SIGN_KEY)
                .compressWith(CompressionCodecs.GZIP)
                .compact();
        return jwtToken;
    }

    //2. 根据token获取用户信息
    public String getUserInfoFormToken(String token) {
        String userinfo = Jwts.parser().setSigningKey(TOKEN_SIGN_KEY).parseClaimsJws(token).getBody().getSubject();
        return userinfo;
    }
}
