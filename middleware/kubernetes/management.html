def createVersion() {
    return new Date().format('yyyyMMddHHmmss') + "_${env.BUILD_ID}"
  }

pipeline {
  agent {
    node {
      label 'maven'
    }

  }
  stages {
    stage('拉取代码') {
      agent none
      steps {
        container('maven') {
          git(url: 'https://codeup.aliyun.com/64d2358dbebb3e7debf55188/be/geo_pleam.git', credentialsId: 'yunxiao-secret', branch: '$Branch', changelog: true, poll: false)
        }

      }
    }

    stage('构建并推送镜像') {
      agent none
      steps {
        container('maven') {
          sh 'mvn clean package -Dmaven.test.skip=true'
          sh 'cd $APP_NAME && docker build  -t $REGISTRY/$RROJECT_NAMESPACE/$APP_NAME:v1-$TAG_NAME .'
          withCredentials([usernamePassword(credentialsId: 'harbor-secret', passwordVariable: 'DOCKER_PASSWORD', usernameVariable: 'DOCKER_USERNAME')]) {
            sh 'echo "$DOCKER_PASSWORD" | docker login $REGISTRY -u "$DOCKER_USERNAME" --password-stdin'
            sh 'docker push $REGISTRY/$RROJECT_NAMESPACE/$APP_NAME:v1-$TAG_NAME'
          }

        }

      }
    }

    stage('部署应用') {
      agent none
      steps {
        container('maven') {
          withCredentials([kubeconfigContent(credentialsId: 'kubeconfig-secret', variable: 'KUBECONFIG_CONFIG')]) {
            sh 'mkdir -p ~/.kube/'
            sh 'echo "$KUBECONFIG_CONFIG" > ~/.kube/config'
            sh 'sed -i "s#IMAGE_NAME#$REGISTRY/$RROJECT_NAMESPACE/$APP_NAME:v1-$TAG_NAME#g" $APP_NAME/deploy/deploy.yaml'
            sh 'envsubst < $APP_NAME/deploy/deploy.yaml | kubectl apply -f -'
          }

        }

      }
    }

  }
  environment {
    KUBECONFIG_CREDENTIAL_ID = 'kubeconfig-secret'
    REGISTRY = 'harbor.prod.geotmt.com:9443'
    RROJECT_NAMESPACE = 'pleam'
    TAG_NAME = createVersion()
  }
}

  agent {
    //kubernetes {

      }
    }

    stage('Run npm install') {
      agent none
      steps {
        container('nodejs') {
          sh 'npm install -g cnpm@6.1.1 --registry=https://registry.npmmirror.com'

      }
    }

    stage('构建并推送镜像') {

11.3.部署dashboard

wget https://oss.linuxtxc.com/deploy/yaml/kubernetes-dashboard.yaml
kubectl apply -f  kubernetes-dashboard.yaml

#查看Nodeport访问端口
kubectl get service -n kubernetes-dashboard
  • https://nodeIP:30081
#创建用户
kubectl create serviceaccount dashboard-admin -n kube-system

#用户授权
kubectl create clusterrolebinding dashboard-admin --clusterrole=cluster-admin --serviceaccount=kube-system:dashboard-admin

#获取token
kubectl describe secrets -n  kube-system $(kubectl -n kube-system get secret | awk '/dashboard-admin/{print $1}')
