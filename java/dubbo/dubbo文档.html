dubbo 使用记录

<!-- 主pom.xml依赖 -->            
<!-- Dubbo -->
<dependencyManagement>
    <dependencies>
        <dependency>
            <groupId>org.springframework.boot</groupId>
            <artifactId>spring-boot-starter-web</artifactId>
            <version>${spring-boot.version}</version>
            <exclusions>
                <exclusion>
                    <groupId>org.springframework.boot</groupId>
                    <artifactId>spring-boot-starter-tomcat</artifactId>
                </exclusion>
            </exclusions>
        </dependency>

        <!-- Dubbo -->
        <dependency>
            <groupId>org.apache.dubbo</groupId>
            <artifactId>dubbo-bom</artifactId>
            <version>3.2.14</version>
            <type>pom</type>
            <scope>import</scope>
        </dependency>
    </dependencies>
</dependencyManagement>

消费者 consumer

<!-- dubbo -->
<dependency>
    <groupId>org.apache.dubbo</groupId>
    <artifactId>dubbo-spring-boot-starter</artifactId>
</dependency>
dubbo:
  application:
    name: ${spring.application.name} #服务名称
  protocol:
    name: dubbo #远程调用协议
    port: -1 #远程调用协议
  registry:
    address: nacos://${spring.cloud.nacos.discovery.server-addr}   #使用nacos作为注册中心
  consumer:
    check: false #启动时不检查 消费者需要的生产者 接口实现
// 启动类上面加上 @EnableDubbo
@SpringBootApplication
@EnableDubbo
public class OpenApiAdminApplication {

	public static void main(String[] args) {
		SpringApplication.run(OpenApiAdminApplication.class, args);
	}

}
// 注入依赖
@DubboReference
private ICustomerViewLogService iCustomerViewLogService;

实际action使用

public class AdminMainTaskAction extends CoreAction {
    
    @DubboReference
	private ICustomerViewLogService iCustomerViewLogService;
    
    @ApiOperation("调用示例")
	@GetMapping("/v1.0/dubbo/get")
	@PermitAll
	public ResultVo get() {
		return success("[{0}]操作执行成功!", new String[] { "查询" }, iCustomerViewLogService.get());
	}
    
}

生产者 provider (例如: circles-provider )

<!-- dubbo -->
<dependency>
    <groupId>org.apache.dubbo</groupId>
    <artifactId>dubbo-spring-boot-starter</artifactId>
</dependency>
dubbo:
  application:
    name: ${spring.application.name} #服务名称
    qos-enable: false # 服务质量保证端口,此处为了防止和本地端口冲突,先关闭
  protocol:
    name: dubbo #远程调用协议
    port: -1 #远程调用端口 -1 代表随机
  registry:
    address: nacos://${spring.cloud.nacos.discovery.server-addr}   #使用nacos作为注册中心
    parameters.register-consumer-url: true #消费者注册到nacos注册中心上
    register-mode: instance/interface/all # 只使用接口级服务发现
  scan:
    base-packages: com.piller.springboot.starter.social.basic # 扫描接口实现类
// 启动类上面加上 @EnableDubbo
@SpringBootApplication
@EnableDubbo
public class SharingCirclesProviderApplication {
    
	public static void main(String[] args) {
		SpringApplication.run(SharingCirclesProviderApplication.class, args);
	}

}
// 注入依赖
@DubboReference
private ICustomerViewLogService iCustomerViewLogService;

实际实现类使用

<!-- 具体 module 的boot也需要添加该依赖 -->
<!-- dubbo -->
<dependency>
    <groupId>org.apache.dubbo</groupId>
    <artifactId>dubbo-spring-boot-starter</artifactId>
</dependency>
// basic 包下面写接口
public interface ICustomerViewLogService {
    String get();
}
    

// 实现类写在具体的 module 里面
@DubboService
public class CustomerViewLogService implements ICustomerViewLogService {

    @Override
    public String get() {
        return "11111111111111111111";
    }
}
