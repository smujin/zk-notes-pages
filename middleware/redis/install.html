安装Redis

1. 安装准备

从Redis中文网 下载最新Redis安装包

# 1.1 下载Redis安装包.
wget https://download.redis.io/releases/redis-7.4.2.tar.gz
# 1.2 解压文件
tar xzf redis-7.4.2.tar.gz
cd redis-7.4.2
# 1.3 编译文件
make
make install

2. 修改配置文件

# 1. 去除IP绑定(需要注释以下行)
bind 127.0.0.1

# 2. 关闭保护模式
protected-mode no

# 3. 开启后台启动
daemonize yes

# 4. 开启 AOF功能(默认是no)
appendonly yes
# AOF文件的名称
appendfilename "appendonly.aof"
# 表示每执行一次写命令，立即记录到AOF文件
appendfsync always 
# 写命令执行完先放入AOF缓冲区，然后表示每隔1秒将缓冲区数据写到AOF文件，是默认方案
appendfsync everysec 
# 写命令执行完先放入AOF缓冲区，由操作系统决定何时将缓冲区内容写回磁盘
appendfsync no

# 5. 设置密码
requirepass 123456

redis 7版本启动时需要打开系统的内存复用，防止内存不够时系统崩溃

vim /etc/sysctl.conf
# 在最后一行加上这个
vm.overcommit_memory = 1
# 最后执行
sysctl vm.overcommit_memory=1
# 重启Redis

3. Redis 命令

# 更改redis版本目录为redis
mv redis-7.4.2 redis

# 1. 启动Redis
redis-server redis.conf

# 2.停止
redis-cli -p 6379 shutdown

# 3.检索服务
ps -ef |grep redis

# 4.杀死进程
kill -9 PID号

# 5.使用客户端
redis-cli -p 6379

# 6.退出客户端
exit

3. 设置Redis 开机自启

#1. 设置开机文件
vim /etc/systemd/system/redis.service

# 写以下内容
[Unit]
Description=redis-server
After=network.target

[Service]
Type=forking
ExecStart=/usr/local/redis/src/redis-server /usr/local/redis/redis.conf
PrivateTmp=true

[Install]
WantedBy=multi-user.target

# 保存系统服务文件，然后输入命令，重载系统服务
systemctl daemon-reload
# 实现开机自启 
systemctl enable redis
# 查看redis 服务的状态
systemctl status redis
