#include <switch.h>
#include <switch_utils.h>
#include <math.h>
#include <string>
#include <memory>
#include <vector>

#include "/usr/src/alibabacloud-nls-cpp-sdk/build/install/NlsSdk3.X_LINUX/include/nlsClient.h"
#include "/usr/src/alibabacloud-nls-cpp-sdk/build/install/NlsSdk3.X_LINUX/include/speechRecognizerRequest.h"
#include "/usr/src/alibabacloud-nls-cpp-sdk/build/install/NlsSdk3.X_LINUX/include/nlsGlobal.h"

using namespace AlibabaNls;

#define SILENCE_THRESHOLD 50
#define MIN_SPEECH_FRAMES 5
#define ENERGY_THRESHOLD 800.0
#define MAX_SENTENCE_FRAMES 800

/* ================= 全局 SDK 线程控制 ================= */

static bool g_sdk_started = false;
static switch_mutex_t* g_sdk_mutex = nullptr;

/* ================= ASR 会话结构 ================= */

struct asr_session {

    switch_core_session_t* session{nullptr};
    switch_media_bug_t* bug{nullptr};
    std::shared_ptr<NlsClient> client;
    SpeechRecognizerRequest* current_request{nullptr};

    std::string token;
    std::string app_key;

    switch_mutex_t* mutex{nullptr};

    switch_mutex_t* recog_mutex{nullptr};
    switch_thread_cond_t* recog_cond{nullptr};
    bool recog_done{false};

    bool running{false};
    bool speaking{false};

    int silence_frames{0};
    int speech_frames{0};
    int voice_frames{0};

    std::vector<uint8_t> audio_buffer;

    asr_session(switch_core_session_t* s) : session(s) {

        switch_memory_pool_t* pool =
            switch_core_session_get_pool(s);

        switch_mutex_init(&mutex,
                          SWITCH_MUTEX_NESTED,
                          pool);

        switch_mutex_init(&recog_mutex,
                          SWITCH_MUTEX_NESTED,
                          pool);

        switch_thread_cond_create(&recog_cond,
                                  pool);

        client = std::shared_ptr<NlsClient>(
            NlsClient::getInstance(),
            [](NlsClient*) {});
    }

    void lock()   { switch_mutex_lock(mutex); }
    void unlock() { switch_mutex_unlock(mutex); }

    /* ================= 回调 ================= */

    static void on_completed(NlsEvent* event, void* userdata) {

        asr_session* sess = (asr_session*)userdata;
        if (!sess || !event) return;

        switch_log_printf(SWITCH_CHANNEL_LOG,
                          SWITCH_LOG_NOTICE,
                          "🎤 句子识别结果: %s\n",
                          event->getAllResponse());

        switch_mutex_lock(sess->recog_mutex);
        sess->recog_done = true;
        switch_thread_cond_signal(sess->recog_cond);
        switch_mutex_unlock(sess->recog_mutex);
    }

    static void on_failed(NlsEvent* event, void* userdata) {

        asr_session* sess = (asr_session*)userdata;
        if (!sess) return;

        switch_log_printf(SWITCH_CHANNEL_LOG,
                          SWITCH_LOG_ERROR,
                          "识别失败: %s\n",
                          event ? event->getErrorMessage() : "unknown");

        switch_mutex_lock(sess->recog_mutex);
        sess->recog_done = true;
        switch_thread_cond_signal(sess->recog_cond);
        switch_mutex_unlock(sess->recog_mutex);
    }

    /* ================= 一句话识别 ================= */

    void recognize_sentence() {

        if (audio_buffer.empty()) {
            switch_log_printf(SWITCH_CHANNEL_LOG,
                            SWITCH_LOG_WARNING,
                            "音频缓冲为空，不识别\n");
            return;
        }

        switch_log_printf(SWITCH_CHANNEL_LOG,
                        SWITCH_LOG_NOTICE,
                        "准备调用阿里一句话识别, 音频大小=%lu\n",
                        (unsigned long)audio_buffer.size());

        auto request = client->createRecognizerRequest();
        if (!request) {
            switch_log_printf(SWITCH_CHANNEL_LOG,
                            SWITCH_LOG_ERROR,
                            "createRecognizerRequest 失败\n");
            return;
        }

        // 准备等待状态
        bool started_ok = false;
        recog_done = false;

        request->setToken(token.c_str());
        if (!app_key.empty())
            request->setAppKey(app_key.c_str());

        request->setFormat("pcm");
        request->setSampleRate(8000); // 你电话场景常用 8k

        request->setPunctuationPrediction(true);
        request->setInverseTextNormalization(true);

        // 回调：start 建立成功
        request->setOnRecognitionStarted([](NlsEvent* event, void* userdata) {
            bool *flag = (bool*)userdata;
            *flag = true;
        }, &started_ok);

        request->setOnRecognitionCompleted(on_completed, this);
        request->setOnTaskFailed(on_failed, this);

        switch_log_printf(SWITCH_CHANNEL_LOG,
                        SWITCH_LOG_NOTICE,
                        "开始 request->start()\n");

        // 发起连接
        int ret = request->start();

        if (ret < 0) {
            switch_log_printf(SWITCH_CHANNEL_LOG,
                            SWITCH_LOG_ERROR,
                            "request->start() 失败 ret=%d\n",
                            ret);

            client->releaseRecognizerRequest(request);
            audio_buffer.clear();
            return;
        }

        // 等待 Websocket 成功（最多 2 秒）
        for (int i = 0; i < 100 && !started_ok; i++) {
            switch_yield(20000); // 等 20ms * 100 = 2s
        }

        if (!started_ok) {
            switch_log_printf(SWITCH_CHANNEL_LOG,
                            SWITCH_LOG_ERROR,
                            "WebSocket 连接未成功，放弃识别\n");
            request->stop();
            client->releaseRecognizerRequest(request);
            audio_buffer.clear();
            return;
        }

        // 正式发送 audio pcm 数据
        switch_log_printf(SWITCH_CHANNEL_LOG,
                        SWITCH_LOG_NOTICE,
                        "WebSocket 连接成功, 发送音频...\n");

        size_t offset = 0;
        size_t frame_size = 1600; // 对 8k PCM，每 1600 字节 ~100ms

        while (offset < audio_buffer.size()) {

            size_t chunk = std::min(frame_size,
                                    audio_buffer.size() - offset);

            int sret = request->sendAudio(
                audio_buffer.data() + offset,
                chunk);

            if (sret < 0) {
                switch_log_printf(SWITCH_CHANNEL_LOG,
                                SWITCH_LOG_ERROR,
                                "sendAudio 失败 ret=%d\n",
                                sret);
                break;
            }

            offset += chunk;

            // 官方推荐: 每发送 frame_size bytes pause Real-Time
            switch_yield(100 * 1000);  // sleep ~100ms
        }

        switch_log_printf(SWITCH_CHANNEL_LOG,
                        SWITCH_LOG_NOTICE,
                        "音频发送完成 调用 stop()\n");

        request->stop();

        // 等待 completed/failed（最多 10 秒）
        switch_mutex_lock(recog_mutex);
        switch_thread_cond_timedwait(
            recog_cond,
            recog_mutex,
            10 * 1000 * 1000);
        switch_mutex_unlock(recog_mutex);

        if (!recog_done) {
            switch_log_printf(SWITCH_CHANNEL_LOG,
                            SWITCH_LOG_ERROR,
                            "等待超时，未收到阿里回调！\n");
        }

        client->releaseRecognizerRequest(request);
        audio_buffer.clear();
    }
};

/* ================= Media Bug ================= */

static switch_bool_t asr_bug_callback(
    switch_media_bug_t* bug,
    void* user_data,
    switch_abc_type_t type)
{
    asr_session* sess =
        (asr_session*)user_data;

    if (!sess) return SWITCH_FALSE;

    if (type == SWITCH_ABC_TYPE_CLOSE) {
        sess->running = false;
        return SWITCH_TRUE;
    }

    if (type != SWITCH_ABC_TYPE_READ_REPLACE ||
        !sess->running)
        return SWITCH_TRUE;

    switch_frame_t* frame =
        switch_core_media_bug_get_read_replace_frame(bug);

    if (!frame || !frame->data ||
        frame->datalen <= 0)
        return SWITCH_TRUE;

    int16_t* samples = (int16_t*)frame->data;
    int sample_count = frame->datalen / 2;

    double sum = 0.0;
    for (int i = 0; i < sample_count; i++) {
        double s = samples[i];
        sum += s * s;
    }

    double rms = sqrt(sum / sample_count);
    bool silence = (rms < ENERGY_THRESHOLD);

    sess->lock();

    if (!silence) {

        sess->silence_frames = 0;
        sess->speech_frames++;

        if (!sess->speaking &&
            sess->speech_frames >= MIN_SPEECH_FRAMES)
        {
            sess->speaking = true;
            sess->voice_frames = 0;

            switch_log_printf(
                SWITCH_CHANNEL_LOG,
                SWITCH_LOG_NOTICE,
                "开始录音一句话\n");
        }

        if (sess->speaking) {

            uint8_t* data =
                (uint8_t*)frame->data;

            sess->audio_buffer.insert(
                sess->audio_buffer.end(),
                data,
                data + frame->datalen);

            sess->voice_frames++;

            if (sess->voice_frames >= MAX_SENTENCE_FRAMES)
            {
                switch_log_printf(
                    SWITCH_CHANNEL_LOG,
                    SWITCH_LOG_NOTICE,
                    "超过最大语音长度，强制识别\n");

                sess->recognize_sentence();

                sess->speaking = false;
                sess->speech_frames = 0;
                sess->silence_frames = 0;
            }
        }

    } else {

        if (sess->speaking) {

            sess->silence_frames++;

            if (sess->silence_frames >= SILENCE_THRESHOLD)
            {
                switch_log_printf(
                    SWITCH_CHANNEL_LOG,
                    SWITCH_LOG_NOTICE,
                    "检测到句尾，开始识别\n");

                sess->recognize_sentence();

                sess->speaking = false;
                sess->speech_frames = 0;
                sess->silence_frames = 0;
            }
        }
    }

    sess->unlock();
    return SWITCH_TRUE;
}

/* ================= start ================= */

SWITCH_STANDARD_APP(start_asr_function)
{
    if (!session || zstr(data)) return;

    char* argv[5] = {0};
    char* mydata = strdup(data);
    int argc = switch_separate_string(
        mydata, ' ', argv, 5);

    if (argc < 1) {
        free(mydata);
        return;
    }

    asr_session* sess =
        new asr_session(session);

    sess->token = argv[0];

    if (argc > 1 && argv[1])
        sess->app_key = argv[1];

    sess->running = true;

    switch_media_bug_t* bug = nullptr;

    if (switch_core_media_bug_add(
            session,
            "asr_sentence",
            NULL,
            asr_bug_callback,
            sess,
            0,
            SMBF_READ_REPLACE,
            &bug)
        != SWITCH_STATUS_SUCCESS)
    {
        delete sess;
        free(mydata);
        return;
    }

    sess->bug = bug;

    switch_channel_set_private(
        switch_core_session_get_channel(session),
        "asr_sentence",
        sess);

    switch_log_printf(
        SWITCH_CHANNEL_LOG,
        SWITCH_LOG_NOTICE,
        "SpeechRecognizer 模式启动\n");

    free(mydata);
}

/* ================= stop ================= */

SWITCH_STANDARD_APP(stop_asr_function)
{
    if (!session) return;

    switch_channel_t* channel =
        switch_core_session_get_channel(session);

    asr_session* sess =
        (asr_session*)switch_channel_get_private(
            channel, "asr_sentence");

    if (!sess) return;

    sess->running = false;

    if (sess->bug)
        switch_core_media_bug_remove(
            session, &sess->bug);

    switch_channel_set_private(
        channel,
        "asr_sentence",
        nullptr);

    delete sess;

    switch_log_printf(
        SWITCH_CHANNEL_LOG,
        SWITCH_LOG_NOTICE,
        "ASR 停止\n");
}

/* ================= module ================= */

SWITCH_MODULE_LOAD_FUNCTION(mod_asr_aliyun_load)
{
    switch_application_interface_t* app;

    *module_interface =
        switch_loadable_module_create_module_interface(
            pool,
            "mod_asr_aliyun");

    switch_mutex_init(&g_sdk_mutex,
                      SWITCH_MUTEX_NESTED,
                      pool);

    switch_mutex_lock(g_sdk_mutex);

    if (!g_sdk_started) {
        NlsClient::getInstance()->startWorkThread(1);
        g_sdk_started = true;

        switch_log_printf(
            SWITCH_CHANNEL_LOG,
            SWITCH_LOG_NOTICE,
            "阿里 SDK 工作线程已启动\n");
    }

    switch_mutex_unlock(g_sdk_mutex);

    SWITCH_ADD_APP(app,
                   "start_asr",
                   "Start ASR",
                   "Start ASR",
                   start_asr_function,
                   "<token> [appkey]",
                   SAF_NONE);

    SWITCH_ADD_APP(app,
                   "stop_asr",
                   "Stop ASR",
                   "Stop ASR",
                   stop_asr_function,
                   "",
                   SAF_NONE);

    switch_log_printf(
        SWITCH_CHANNEL_LOG,
        SWITCH_LOG_NOTICE,
        "mod_asr_aliyun loaded\n");

    return SWITCH_STATUS_SUCCESS;
}

// 模块定义
SWITCH_MODULE_DEFINITION(
    mod_asr_aliyun,
    mod_asr_aliyun_load,
    NULL,
    NULL);
